# movie_scraper.py
import requests
from bs4 import BeautifulSoup
import json
import time

class MovieScraper:
    def __init__(self, base_url):
        self.base_url = base_url
        self.headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
        }
    
    def get_movie_list(self):
        response = requests.get(self.base_url, headers=self.headers)
        soup = BeautifulSoup(response.content, 'html.parser')
        movies = []
        
        movie_items = soup.find_all('div', class_='movie-item')
        for item in movie_items:
            title = item.find('h3').text.strip() if item.find('h3') else None
            link = item.find('a')['href'] if item.find('a') else None
            rating = item.find('.rating').text.strip() if item.find('.rating') else None
            year = item.find('.year').text.strip() if item.find('.year') else None
            
            if title and link:
                movies.append({
                    'title': title,
                    'link': link,
                    'rating': rating,
                    'year': year
                })
        
        return movies
    
    def get_movie_details(self, movie_url):
        response = requests.get(movie_url, headers=self.headers)
        soup = BeautifulSoup(response.content, 'html.parser')
        
        # Extract details based on the actual structure
        synopsis = soup.find('.synopsis').text.strip() if soup.find('.synopsis') else None
        director = soup.find('.director').text.strip() if soup.find('.director') else None
        cast = [item.text.strip() for item in soup.find_all('.cast-item')]
        
        return {
            'synopsis': synopsis,
            'director': director,
            'cast': cast
        }
    
    def scrape_all(self):
        movies = self.get_movie_list()
        for movie in movies:
            details = self.get_movie_details(movie['link'])
            movie.update(details)
            time.sleep(1)  # Rate limiting
        
        return movies

# Usage
if __name__ == "__main__":
    scraper = MovieScraper("http://45.11.57.188/")
    all_movies = scraper.scrape_all()
    
    with open('movies.json', 'w') as f:
        json.dump(all_movies, f)

# process_data.py
import json
import sqlite3

def create_database():
    conn = sqlite3.connect('movies.db')
    cursor = conn.cursor()
    
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS movies (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL,
        slug TEXT UNIQUE,
        synopsis TEXT,
        director TEXT,
        rating TEXT,
        year TEXT,
        link TEXT,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
    ''')
    
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS cast (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        movie_id INTEGER,
        name TEXT,
        FOREIGN KEY (movie_id) REFERENCES movies (id)
    )
    ''')
    
    conn.commit()
    return conn

def import_movies(conn, movies_file):
    with open(movies_file, 'r') as f:
        movies = json.load(f)
    
    cursor = conn.cursor()
    for movie in movies:
        cursor.execute('''
        INSERT OR REPLACE INTO movies 
        (title, slug, synopsis, director, rating, year, link)
        VALUES (?, ?, ?, ?, ?, ?, ?)
        ''', (
            movie.get('title'),
            movie.get('title', '').lower().replace(' ', '-'),
            movie.get('synopsis'),
            movie.get('director'),
            movie.get('rating'),
            movie.get('year'),
            movie.get('link')
        ))
        
        movie_id = cursor.lastrowid
        for actor in movie.get('cast', []):
            cursor.execute('INSERT INTO cast (movie_id, name) VALUES (?, ?)', (movie_id, actor))
    
    conn.commit()

if __name__ == "__main__":
    conn = create_database()
    import_movies(conn, 'movies.json')
    conn.close()
